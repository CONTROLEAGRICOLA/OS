// SOLICITA PERMISSÃO DE NOTIFICAÇÃO AO ENTRAR window.addEventListener('DOMContentLoaded', async () => { carregarPreferenciaTema(); verificarSessaoSalva(); await carregarDadosIniciais(); if ("Notification" in window && Notification.permission !== "granted") { Notification.requestPermission(); } // Monitora atualizações de status para emitir notificação a cada 20 segundos setInterval(verificarAtualizacaoStatusNotificacoes, 20000); }); // MONITOR DE NOTIFICAÇÃO PUSH DE APROVAÇÃO/REJEIÇÃO async function verificarAtualizacaoStatusNotificacoes() { if (!solicitanteCracha || !navigator.onLine) return; try { const res = await fetch(`${APPSCRIPT_URL}?action=buscar_os&solicitante=${encodeURIComponent(solicitanteCracha)}`); const data = await res.json(); if (data.success && data.dados) { const statusSalvos = JSON.parse(localStorage.getItem('os_tracker_status') || '{}'); data.dados.forEach(os => { const statusAnterior = statusSalvos[os.codigoOS]; const statusAtual = os.statusAprovacao; if (statusAnterior && statusAnterior === 'Em análise' && statusAtual !== 'Em análise') { dispararNotificacaoNavegador(os); } statusSalvos[os.codigoOS] = statusAtual; }); localStorage.setItem('os_tracker_status', JSON.stringify(statusSalvos)); } } catch (e) { console.warn('Erro ao checar notificações.'); } } function dispararNotificacaoNavegador(os) { if ("Notification" in window && Notification.permission === "granted") { const aprovado = os.statusAprovacao === 'Aprovado'; const titulo = aprovado ? `✅ OS nº ${os.codigoOS} APROVADA!` : `❌ OS nº ${os.codigoOS} REJEITADA!`; const corpo = aprovado ? `Sua solicitação para a fazenda ${os.descricao_fazenda} foi aprovada pelo Controle Agrícola.` : `Motivo: ${os.justificativaRejeicao || 'Não especificado.'}`; new Notification(titulo, { body: corpo, icon: 'logo.png' }); } } // RENDERIZADOR ATUALIZADO NA TABELA DA ABA GESTÃO OS DO APP function renderizarTabelaGestao(lista) { const tbody = document.getElementById('gestao-tbody'); tbody.innerHTML = ''; if (lista.length === 0) { tbody.innerHTML = 'Nenhuma OS localizada para este crachá.'; return; } lista.forEach((os, index) => { const tr = document.createElement('tr'); tr.className = 'tr-clickable'; tr.onclick = () => visualizarOS(index); const st = os.statusAprovacao || "Em análise"; let statusBadge = `⏳ Em análise`; if (st === "Aprovado") statusBadge = `🟢 Aprovado`; if (st === "Rejeitado") statusBadge = `🔴 Rejeitado`; tr.innerHTML = ` #${os.codigoOS || 'PENDENTE'} ${os.data_solicitacao || '-'} ${os.fa_codigo || ''} - ${os.descricao_fazenda || ''} ${os.operacao || '-'} ${os.talhoes || '-'} ${os.equipamentos || os.frota || '-'} ${statusBadge} `; tbody.appendChild(tr); }); }
